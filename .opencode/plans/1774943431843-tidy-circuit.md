# Chat Events Refactor & Presence/Typing APIs

## Summary

1. Create const object for event names, derive enums from it
2. Replace string literals throughout app with const references
3. Move message persistence to WebSocket `send_message` handler
4. Add REST APIs for presence and typing state

### File: `apps/api/src/app/chat/chat-events.ts`

Create const object and derive enums:

```typescript
import { z } from "zod";

// Event name constants
export const CHAT_EVENTS = {
  // Client → Server
  JOIN_ROOM: "join_room",
  LEAVE_ROOM: "leave_room",
  SEND_MESSAGE: "send_message",
  TYPING_START: "typing_start",
  TYPING_STOP: "typing_stop",
  HEARTBEAT: "heartbeat",
  // Server → Client
  MESSAGE_NEW: "message:new",
  USER_TYPING: "user:typing",
  USER_PRESENCE: "user:presence",
  MEMBER_JOINED: "member:joined",
  MEMBER_LEFT: "member:left",
  ROOM_DELETED: "room:deleted",
} as const;

// Derive enum types from const object values
export const clientToServerEventNameEnum = z.enum([
  CHAT_EVENTS.JOIN_ROOM,
  CHAT_EVENTS.LEAVE_ROOM,
  CHAT_EVENTS.SEND_MESSAGE,
  CHAT_EVENTS.TYPING_START,
  CHAT_EVENTS.TYPING_STOP,
  CHAT_EVENTS.HEARTBEAT,
]);

export const serverToClientEventNameEnum = z.enum([
  CHAT_EVENTS.MESSAGE_NEW,
  CHAT_EVENTS.USER_TYPING,
  CHAT_EVENTS.USER_PRESENCE,
  CHAT_EVENTS.MEMBER_JOINED,
  CHAT_EVENTS.MEMBER_LEFT,
  CHAT_EVENTS.ROOM_DELETED,
```

## Phase 3: Move Message Persistence to WebSocket

### File: `apps/api/src/app/chat/chat.gateway.ts`

Update `send_message` handler to persist via ChatService:

```typescript
@SubscribeMessage(CHAT_EVENTS.SEND_MESSAGE)
async handleSendMessage(
  @ConnectedSocket() client: AuthenticatedSocket,
  @MessageBody() payload: unknown,
) {
  const parsed = wsSendMessageInputSchema.safeParse(payload);
  if (!parsed.success) {
    client.emit("error", { message: "Invalid payload" });
    return;
  }

  const { roomId, content } = parsed.data;
  const userId = client.data.userId;

  // Persist message via ChatService
  const message = await this.chatService.createMessage(
    userId,
    roomId,
    { content },
  );

  // Broadcast to room
  client.to(`room:${roomId}`).emit(CHAT_EVENTS.MESSAGE_NEW, {
    roomId,
    senderId: userId,
    senderName: client.data.userName,
    content: message.content,
    createdAt: message.createdAt.toISOString(),
  });
}
```

```typescript
async getPresence(roomId: string): Promise<{ onlineUsers: Array<{ userId: string; userName: string }> }> {
  const userIds = await this.presenceService.getOnlineUsers(roomId);
  // Fetch user details from DB for each userId
  const users = await this.drizzle.db
    .select({ id: schema.user.id, name: schema.user.name })
    .from(schema.user)
    .where(inArray(schema.user.id, userIds));

  return { onlineUsers: users.map(u => ({ userId: u.id, userName: u.name })) };
}

async getTyping(roomId: string): Promise<{ typingUsers: Array<{ userId: string; userName: string }> }> {
  const userIds = await this.typingService.getTypingUsers(roomId);
  if (userIds.length === 0) {
    return { typingUsers: [] };
  }

  const users = await this.drizzle.db
    .select({ id: schema.user.id, name: schema.user.name })
    .from(schema.user)
    .where(inArray(schema.user.id, userIds));

  return { typingUsers: users.map(u => ({ userId: u.id, userName: u.name })) };
}
```
