# AI Copilot Integration Plan

## Overview

Add an AI assistant ("Copilot") to the chat that can be invoked via `@copilot` mentions. Copilot responds asynchronously via a background queue processor.

## Design Decisions

| Decision           | Choice                                           |
| ------------------ | ------------------------------------------------ |
| @ Dropdown         | Copilot only (extensible later)                  |
| Mention display    | Keep `@copilot` visible in message               |
| Auto-join          | Copilot added to every new room automatically    |
| Context management | Sliding window (drop oldest), use tiktoken       |
| Context limit      | 80% of model context window                      |
| Model              | gpt-5-mini                                       |
| System prompt      | Hardcoded, instructions only                     |
| AI failures        | Retry 3x with backoff, then post error message   |
| Copilot user       | Seeded bot user in DB                            |
| Bot identification | New `isBot` column on user table                 |
| Rate limiting      | 5 messages per minute per user                   |
| Copilot identity   | Name: "Copilot", Email: "copilot@converge.local" |

### 2.2 Copilot Processor

**File**: `apps/api/src/app/chat/copilot/copilot.processor.ts` (new)

```typescript
@Injectable()
@Processor(COPILOT_QUEUE)
export class CopilotProcessor extends WorkerHost {
  constructor(
    private readonly chatService: ChatService,
    private readonly chatGateway: ChatGateway,
    private readonly aiService: CopilotAiService,
    private readonly drizzle: Drizzle,
  ) {
    super();
  }
  async process(job: Job<CopilotMessageJob>) {
    const { messageId, roomId, senderId, content } = job.data;

    try {
      // 1. Get context (messages up to80% of token limit)
      const context = await this.aiService.buildContext(roomId);

      // 2. Check if @copilot is mentioned
      if (!content.toLowerCase().includes("@copilot")) {
        return; // No AI response needed
      }

      // 3. Call AI
      const response = await this.aiService.generateResponse(context, content);

      // 4. Create copilot message
      const copilotMessage = await this.chatService.createMessage(
        COPILOT_USER_ID,
        roomId,
        { content: response },
      );

      // 5. Broadcast to room
      this.chatGateway.server
        .to(`room:${roomId}`)
        .emit(CHAT_EVENTS.MESSAGE_NEW, {
          id: copilotMessage.id,
          roomId,
          senderId: COPILOT_USER_ID,
          senderName: "Copilot",
          senderEmail: "copilot@converge.local",
          content: response,
          createdAt: copilotMessage.createdAt,
        });
    } catch (error) {
      // Final failure after retries - post error message
      await this.handleFailure(roomId, error);
    }
  }
}
```

import { encoding_for_model, type Tiktoken } from "tiktoken";

export class TokenCounter {
  private encoder: Tiktoken;
  constructor() {
    this.encoder = encoding_for_model("gpt-4o-mini"); // or gpt-5-mini when available
  }
  countTokens(text: string): number {
    return this.encoder.encode(text).length;
  }

  countMessagesTokens(messages: Message[]): number {
    return messages.reduce((sum, msg) => {
      return sum + this.countTokens(msg.content) + 4; // ~4 tokens per message overhead
    }, 0);
  }
}

### 3.2 Context Builder Service

**File**: `apps/api/src/app/chat/copilot/copilot-ai.service.ts` (new)

```typescript
@Injectable()
export class CopilotAiService {
  private readonly tokenCounter = new TokenCounter();
  private readonly MODEL_CONTEXT_LIMIT = 128_000; // gpt-4o-mini context
  private readonly CONTEXT_PERCENTAGE = 0.8; // 80%
  private readonly MAX_CONTEXT_TOKENS = Math.floor(
    this.MODEL_CONTEXT_LIMIT * this.CONTEXT_PERCENTAGE,
  );

  async buildContext(roomId: string): Promise<Message[]> {
    // Fetch all messages for room, ordered by createdAt DESC
    const messages = await this.drizzle.db
      .select()
      .from(chatMessage)
      .where(eq(chatMessage.roomId, roomId))
      .orderBy(desc(chatMessage.createdAt));

    // Sliding window: include messages until we hit token limit
    const context: Message[] = [];
    let totalTokens = 0;

    for (const msg of messages) {
      const msgTokens = this.tokenCounter.countTokens(msg.content) + 4;
      if (totalTokens + msgTokens > this.MAX_CONTEXT_TOKENS) {
        break; // Drop oldest messages
      }
      context.unshift(msg); // Add to beginning (oldest first)
      totalTokens += msgTokens;
    }

    return context;
  }

  async generateResponse(
    context: Message[],
    userMessage: string,
  ): Promise<string> {
    const messages: CoreMessage[] = [
      {
        role: "system",
        content: SYSTEM_PROMPT,
      },
      ...context.map((msg) => ({
        role:
          msg.senderId === COPILOT_USER_ID ? "assistant" : ("user" as const),
        content: msg.content,
      })),
      {
        role: "user",
        content: userMessage,
      },
    ];

    const result = await generateText({
      model: models.chat,
      messages,
    });

    return result.text;
  }
}
```

## Phase 4: Backend - Rate Limiting

### 4.1 Rate Limiter Guard

**File**: `apps/api/src/app/chat/guards/copilot-rate-limit.guard.ts` (new)

```typescript
@Injectable()
export class CopilotRateLimitGuard {
  private readonly LIMIT = 5; // 5 messages
  private readonly WINDOW = 60_000; // 1 minute

  constructor(private readonly redis: RedisService) {}

  async checkLimit(userId: string): Promise<boolean> {
    const key = `copilot:rate:${userId}`;
    const count = await this.redis.incr(key);

    if (count === 1) {
      await this.redis.expire(key, this.WINDOW / 1000);
    }

    return count <= this.LIMIT;
  }
}
```

## Phase 8: BullMQ Job Events for WebSocket Emit

**Problem**: `CopilotProcessor` runs in worker process where `ChatGateway.server` is undefined.

**Solution**: Use `QueueEventsListener` in main app to listen for job completions and emit via WebSocket.

### 8.1 Create CopilotQueueEvents in App Module

**File**: `apps/api/src/app/chat/copilot-queue.events.ts` (new)

```typescript
import {
  OnQueueEvent,
  QueueEventsHost,
  QueueEventsListener,
} from "@nestjs/bullmq";
import { Injectable, Logger } from "@nestjs/common";
import { ChatGateway } from "./chat.gateway.js";
import { COPILOT_QUEUE } from "./copilot/types.js";
import { schema } from "@repo/database";
import { DrizzleService } from "#src/modules/drizzle/drizzle.service.js";
import { COPILOT_USER_ID } from "@repo/database/constants.js";

@Injectable()
@QueueEventsListener(COPILOT_QUEUE)
export class CopilotQueueEvents extends QueueEventsHost {
  private readonly logger = new Logger(CopilotQueueEvents.name);

  constructor(
    private readonly chatGateway: ChatGateway,
    private readonly drizzle: DrizzleService,
  ) {
    super();
  }

  @OnQueueEvent("completed")
  async onCompleted(
    args: { jobId: string; returnvalue: string | null; prev?: string },
    id: string,
  ) {
    this.logger.log(`Copilot job ${args.jobId} completed`);

    // args.returnvalue contains the messageId we returned from processor
    if (!args.returnvalue) return;

    const messageId = args.returnvalue;
    const message = await this.drizzle.db.query.chatMessage.findFirst({
      where: eq(schema.chatMessage.id, messageId),
    });

    if (message && this.chatGateway.server) {
      this.chatGateway.server.to(`room:${message.roomId}`).emit("message:new", {
        id: message.id,
        roomId: message.roomId,
        senderId: COPILOT_USER_ID,
        senderName: "Copilot",
        senderEmail: "copilot@converge.local",
        content: message.content,
        createdAt: message.createdAt.toISOString(),
      });
    }
  }

  @OnQueueEvent("failed")
  async onFailed(
    args: { jobId: string; failedReason: string; prev?: string },
    id: string,
  ) {
    this.logger.error(`Copilot job ${args.jobId} failed: ${args.failedReason}`);
  }
}
```

### 9.3 Context Performance

**Problem**: `buildContext` fetches ALL messages, then uses O(n²) unshift. Slow as chat grows.

**Solution**:

1. Add DB limit + index
2. Push + reverse (O(n) instead of O(n²))
3. Cache context in Redis

**File**: `apps/api/src/app/chat/copilot/copilot-ai.service.ts`

```typescript
async buildContext(roomId: string): Promise<ContextMessage[]> {
  // Estimate ~500 max messages needed for context window
  const MAX_FETCH = 500;

  const messages = await this.drizzle.db
    .select({
      senderId: schema.chatMessage.senderId,
      content: schema.chatMessage.content,
      createdAt: schema.chatMessage.createdAt,
    })
    .from(schema.chatMessage)
    .where(eq(schema.chatMessage.roomId, roomId))
    .orderBy(desc(schema.chatMessage.createdAt))
    .limit(MAX_FETCH);  // Cap DB query

  const context: ContextMessage[] = [];
  let totalTokens = 0;

  // Use push + reverse instead of unshift (O(n) vs O(n²))
  for (const msg of messages) {
    const msgTokens = this.tokenCounter.countTokens(msg.content) + 4;
    if (totalTokens + msgTokens > COPILOT_MAX_CONTEXT_TOKENS) break;
    context.push(msg);  // O(1)
    totalTokens += msgTokens;
  }

  return context.reverse();  // O(n) once
}
```

import { Injectable, Logger } from "@nestjs/common";
import { modelSchema, type ModelSchema } from "./models.schema.js";

const MODELS_DEV_URL = "https://models.dev/api.json";
const CACHE_TTL_MS = 24 * 60 * 60 * 1000; // 24 hours

@Injectable()
export class ModelInfoService {
  private readonly logger = new Logger(ModelInfoService.name);
  private cache: Map<string, ModelSchema> = new Map();
  private lastFetch: number = 0;

  async getContextLimit(modelId: string): Promise<number> {
    const models = await this.fetchModels();
    const model = models[modelId];
    return model?.limit?.context ?? 128_000; // fallback to gpt-4o-mini default
  }

  private async fetchModels(): Promise<Record<string, ModelSchema>> {
    // Check if cache is fresh
    if (this.cache.size > 0 && Date.now() - this.lastFetch < CACHE_TTL_MS) {
      return Object.fromEntries(this.cache);
    }

    try {
      const res = await fetch(MODELS_DEV_URL);
      const data = await res.json();

      // Validate with zod
      const models: Record<string, ModelSchema> = {};
      for (const [id, model] of Object.entries(data)) {
        const parsed = modelSchema.safeParse(model);
        if (parsed.success) {
          models[id] = parsed.data;
          this.cache.set(id, parsed.data);
        }
      }

      this.lastFetch = Date.now();
      this.logger.log(
        `Fetched ${Object.keys(models).length} models from models.dev`,
      );
      return models;
    } catch (error) {
      this.logger.error("Failed to fetch models", error);
      // Return cached data if available
      return Object.fromEntries(this.cache);
    }
  }
}

## Phase 10: Additional Improvements

### 10.1 Redis Caching for Model Info

**Problem**: In-memory cache doesn't survive restarts and doesn't share across instances.

**Solution**: Use Redis for distributed caching.

**File**: `apps/api/src/app/chat/copilot/model-info.service.ts`

```typescript
import { Injectable, Logger } from "@nestjs/common";
import { RedisService } from "#src/modules/redis/redis.service.js";
import { modelSchema, type ModelSchema } from "./models.schema.js";

const MODELS_DEV_URL = "https://models.dev/api.json";
const CACHE_KEY = "copilot:models";
const CACHE_TTL_MS = 24 * 60 * 60 * 1000; // 24 hours

@Injectable()
export class ModelInfoService {
  private readonly logger = new Logger(ModelInfoService.name);

  constructor(private readonly redis: RedisService) {}

  async getContextLimit(modelId: string): Promise<number> {
    const models = await this.fetchModels();
    const model = models[modelId];
    return model?.limit?.context ?? 128_000;
  }

  async getModel(modelId: string): Promise<ModelSchema | undefined> {
    const models = await this.fetchModels();
    return models[modelId];
  }

  private async fetchModels(): Promise<Record<string, ModelSchema>> {
    // Try cache first
    const cached = await this.redis.get<Record<string, ModelSchema>>(CACHE_KEY);
    if (cached) {
      this.logger.debug("Using cached model info from Redis");
      return cached;
    }

    // Fetch from models.dev
    try {
      const res = await fetch(MODELS_DEV_URL);
      const data = await res.json();

      const models: Record<string, ModelSchema> = {};
      for (const [id, model] of Object.entries(
        data as Record<string, unknown>,
      )) {
        const parsed = modelSchema.safeParse(model);
        if (parsed.success) {
          models[id] = parsed.data;
        }
      }

      // Cache in Redis
      await this.redis.set(CACHE_KEY, models, { ttl: CACHE_TTL_MS });
      this.logger.log(
        `Fetched and cached ${Object.keys(models).length} models`,
      );
      return models;
    } catch (error) {
      this.logger.error("Failed to fetch models", error);
      return {};
    }
  }
}
```
